  set(LWS_WITH_SHARED 0)

  set(LWS_WITH_SHARED 1)
endif()

# USE_OPENSSL is passed in as "the backend presents the OpenSSL API", so it is
# already ON for AWS-LC. libwebsockets needs no AWS-LC awareness: it consumes
# libssl/libcrypto and openssl/*.h, all of which AWS-LC provides at the same
# paths. Its LWS_HAVE_* feature flags below (HMAC_CTX_new,
# SSL_EXTRA_CHAIN_CERTS, OPENSSL_ECDH_H, EVP_MD_CTX_free) are all satisfied by
# AWS-LC, so the hardcoded =1 values remain correct.
if (USE_OPENSSL)
  set(LWS_WITH_MBEDTLS OFF)
else()
  set(LWS_WITH_MBEDTLS ON)
endif()

if (WIN32)
  set(LWS_WITH_THREADPOOL ${LWS_WITH_THREADPOOL})
  set(LWS_EXT_PTHREAD_INCLUDE_DIR ${LWS_EXT_PTHREAD_INCLUDE_DIR})
  set(LWS_EXT_PTHREAD_LIBRARIES ${LWS_EXT_PTHREAD_LIBRARIES})
  set(LWS_OPENSSL_INCLUDE_DIRS ${LWS_OPENSSL_INCLUDE_DIRS})
  set(LWS_OPENSSL_LIBRARIES ${LWS_OPENSSL_LIBRARIES})
endif()

# Force off libwebsockets' pipe2() detection on macOS.
#
# lws detects pipe2 with CHECK_C_SOURCE_COMPILES. Starting with the macOS 26
# SDK, <sys/unistd.h> *declares* pipe2 annotated as "introduced in macOS 27.0",
# so on a macOS 26 deployment target the probe compiles with only an
# -Wunguarded-availability-new warning and lws concludes the platform has it.
# libSystem on macOS 26 does not export pipe2, so the call site is weakly bound
# to NULL and lws_plat_pipe_create() jumps to address 0 -- lws_create_context()
# segfaults before any SDK code runs.
#
# Pre-seeding the cache variable makes CHECK_C_SOURCE_COMPILES skip its probe,
# so lws_config.h gets /* #undef LWS_HAVE_PIPE2 */ and lws falls back to plain
# pipe(), which macOS has always provided. Unconditional for Apple: no macOS
# release exports pipe2 today, and if a future one does, the SDK's deployment
# target has to be raised to match before it would be safe to use anyway.
#
# Unrelated to the crypto backend -- this bites OpenSSL, mbedTLS and AWS-LC
# builds identically.
if (APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  set(LWS_PLATFORM_ARGS -DLWS_HAVE_PIPE2=0)
else()
  set(LWS_PLATFORM_ARGS "")
endif()

# libwebsockets revision. mbedTLS 4 needs v5.0.0 (first release tag carrying the
# mbedtls4 migration); every other build (OpenSSL, mbedTLS 3.x) keeps the v4.3.10
# release tag it has always used.
if (USE_MBEDTLS4)
    # v5.0.0 also carries the parallel-connect fds cleanup fix (upstream #3633),
    # so no build-time patch is needed anymore. "README.md" skips cloning the
    # qifs submodule (QUIC test data; unused, QUIC/HTTP3 are off below).
    set(LWS_GIT_TAG        v5.0.0)
    set(LWS_GIT_SHALLOW    TRUE)
    set(LWS_GIT_SUBMODULES "README.md")
    set(LWS_PATCH_ARGS     "")
else()
    set(LWS_GIT_TAG        v4.3.10)
    set(LWS_GIT_SHALLOW    TRUE)
    set(LWS_GIT_SUBMODULES "")
    set(LWS_PATCH_ARGS     "")
    # Test only: compile out the foreign-thread pollfd fallback, as v5.0.0 does on FreeRTOS and
    # Windows. Lets CI catch a signaling wake that only works because of that fallback.
    if (KVS_TEST_LWS_NO_FOREIGN_THREAD_POLLFD)
        set(LWS_PATCH_ARGS PATCH_COMMAND git apply --ignore-whitespace ${CMAKE_CURRENT_LIST_DIR}/libwebsockets-test-no-foreign-thread-pollfd.patch)
    endif()
endif()

    GIT_PROGRESS      TRUE
    GIT_SHALLOW       ${LWS_GIT_SHALLOW}
    GIT_SUBMODULES    "${LWS_GIT_SUBMODULES}"
    ${LWS_PATCH_ARGS}
    PREFIX            ${CMAKE_CURRENT_BINARY_DIR}/build
    LIST_SEPARATOR    |
    CMAKE_ARGS
        ${LWS_PLATFORM_ARGS}
        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
        -DCMAKE_C_FLAGS=${CMAKE_C_FLAGS}
        -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
        -DOPENSSL_ROOT_DIR=${OPENSSL_DIR}
        -DDISABLE_WERROR=ON
        -DLWS_WITH_HTTP2=1
        -DLWS_ROLE_QUIC=OFF
        -DLWS_WITH_HTTP3=OFF
        -DLWS_HAVE_HMAC_CTX_new=1
        -DLWS_HAVE_SSL_EXTRA_CHAIN_CERTS=1
        -DLWS_HAVE_OPENSSL_ECDH_H=1
        -DLWS_HAVE_EVP_MD_CTX_free=1
        -DLWS_WITHOUT_SERVER=1
        -DLWS_WITHOUT_TESTAPPS=1
        -DLWS_WITHOUT_TEST_SERVER_EXTPOLL=1
        -DLWS_WITHOUT_TEST_PING=1
        -DLWS_WITHOUT_TEST_CLIENT=1
        -DLWS_STATIC_PIC=1
        -DLWS_WITH_STATIC=1
        -DLWS_WITH_SHARED=${LWS_WITH_SHARED}
        -DLWS_WITH_MBEDTLS=${LWS_WITH_MBEDTLS}
        -DLWS_MBEDTLS_LIBRARIES=${LWS_MBEDTLS_LIBRARIES}
        -DLWS_MBEDTLS_INCLUDE_DIRS=${LWS_MBEDTLS_INCLUDE_DIRS}
        -DLWS_WITH_MINIMAL_EXAMPLES=0
        -DLWS_HAVE_PTHREAD_H=1
        -DLWS_WITH_THREADPOOL=${LWS_WITH_THREADPOOL}
        -DLWS_WITH_ZLIB=0
        -DLWS_HAVE_EVENTFD=0
        -DLWS_EXT_PTHREAD_INCLUDE_DIR=${LWS_EXT_PTHREAD_INCLUDE_DIR}
        -DLWS_EXT_PTHREAD_LIBRARIES=${LWS_EXT_PTHREAD_LIBRARIES}
        -DLWS_OPENSSL_INCLUDE_DIRS=${LWS_OPENSSL_INCLUDE_DIRS}
        -DLWS_OPENSSL_LIBRARIES=${LWS_OPENSSL_LIBRARIES}
        -DLWS_WITH_IPV6=ON
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      ""
)
