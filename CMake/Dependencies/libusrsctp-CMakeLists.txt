cmake_minimum_required(VERSION 3.6.3)

project(libusrsctp-download NONE)

include(ExternalProject)

# Set by build_dependency() when CMake 4 needs it
if(DEFINED CMAKE_POLICY_VERSION_MINIMUM)
  set(POLICY_VERSION_ARGS -DCMAKE_POLICY_VERSION_MINIMUM=${CMAKE_POLICY_VERSION_MINIMUM})
endif()

ExternalProject_Add(project_libusrsctp
    GIT_REPOSITORY    https://github.com/sctplab/usrsctp.git
    GIT_TAG           c625e21950600460cb5c6075616eb1622df2b06a
    GIT_PROGRESS      TRUE
    PREFIX            ${CMAKE_CURRENT_BINARY_DIR}/build
    CMAKE_ARGS        -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
                      -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
                      ${POLICY_VERSION_ARGS}
                      "-DCMAKE_C_FLAGS=${CMAKE_C_FLAGS} -fPIC"
                      -Dsctp_werror=0
                      -Dsctp_build_programs=0
    BUILD_ALWAYS      TRUE
    TEST_COMMAND      ""
)
