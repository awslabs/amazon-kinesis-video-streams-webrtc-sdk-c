cmake_minimum_required(VERSION 3.6.3)

project(libmbedtls-download NONE)

include(ExternalProject)

if (BUILD_OLD_MBEDTLS_VERSION)
  SET(MBEDTLS_GIT_TAG "v2.28.8")
elseif (USE_MBEDTLS4)
  # mbedTLS 4.x ships PSA Crypto as a separate library (libtfpsacrypto) and is
  # the line tracked for ESP-IDF v6.0 and other PSA-Crypto-first consumers.
  # v4.1.0 matches the version libwebsockets's mbedtls4 support was validated
  # against (warmcat/libwebsockets `3079d31`).
  SET(MBEDTLS_GIT_TAG "v4.1.0")
else()
  SET(MBEDTLS_GIT_TAG "v3.6.6")
endif()

if (BUILD_STATIC_LIBS)
  set(BUILD_SHARED 0)
else()
  set(BUILD_SHARED 1)
endif()

if (${CMAKE_SYSTEM_NAME} MATCHES "Darwin")
  set(CMAKE_MACOSX_RPATH 1)
else()
  set(CMAKE_MACOSX_RPATH 0)
endif()

message(STATUS "C flags here are ${CMAKE_C_FLAGS}")
message(STATUS "Building mbedTLS tag ${MBEDTLS_GIT_TAG}")

# mbedTLS 4 ships PSA Crypto as a static library (libtfpsacrypto). When
# kvsWebrtcClient is built as a shared library we still need that static lib
# to be position-independent so the final shared link succeeds. Same trick
# we used for the libsrtp 3.x build.
set(MBEDTLS_EXTRA_CMAKE_ARGS "")
if (USE_MBEDTLS4)
  list(APPEND MBEDTLS_EXTRA_CMAKE_ARGS -DCMAKE_POSITION_INDEPENDENT_CODE=ON)
endif()

ExternalProject_Add(
  project_libmbedtls
  GIT_REPOSITORY  https://github.com/Mbed-TLS/mbedtls.git
  GIT_TAG         ${MBEDTLS_GIT_TAG}
  GIT_PROGRESS    TRUE
  GIT_SHALLOW     TRUE
  PREFIX          ${CMAKE_CURRENT_BINARY_DIR}/build
  CMAKE_ARGS
    -DCMAKE_INSTALL_PREFIX=${OPEN_SRC_INSTALL_PREFIX}
    -DUSE_SHARED_MBEDTLS_LIBRARY=${BUILD_SHARED}
    -DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}
    -DCMAKE_MACOSX_RPATH=${CMAKE_MACOSX_RPATH}
    -DENABLE_TESTING=OFF
    -DENABLE_PROGRAMS=OFF
    ${MBEDTLS_EXTRA_CMAKE_ARGS}
    "-DCMAKE_C_FLAGS=${CMAKE_C_FLAGS} -Wno-array-bounds"
  BUILD_ALWAYS    TRUE
  TEST_COMMAND    "")
